"""A validated zero-step seed is the origin of its moving-metric chart."""
import jax
import jax.numpy as jnp
import numpy as np

from gradscf import scf
from gradscf.scf import orbital_optimization as orbital


def test_zero_step_preserves_a_stationary_seed_with_metric_roundoff():
    # Within the existing 1e-8 orthonormal-input contract. Re-normalization
    # changes the density and hence the physical Fock at this strict root.
    coeff = jnp.diag(jnp.array([1.+5e-10, 1.]))
    factor = jnp.array([[.5, .2], [.2, .1]])
    eri = jnp.einsum("pq,rs->pqrs", factor, factor)
    h = jnp.array([[-1., -.1*coeff[0, 0]**2], [-.1*coeff[0, 0]**2, .5]])
    state = scf.minimize_roks_from_integrals(
        overlap=jnp.eye(2), hcore=h, eri=eri, nuclear_repulsion=0.,
        ao=jnp.zeros((0, 2)), ao_deriv1=jnp.zeros((4, 0, 2)), grid_weights=jnp.zeros(0),
        mo_coeff=coeff, mo_occ=np.array([[1., 0.], [1., 0.]]),
        max_iterations=0, orthonormalize_initial=False, gradient_tolerance=1e-11)
    assert bool(state.stationary)
    np.testing.assert_allclose(state.mo_coeff, coeff, atol=2e-15, rtol=0)


def test_reference_metric_transport_is_identity_but_keeps_pulay_response():
    metric = jnp.array([[1.2, .1], [.1, .8]])
    coeff = orbital._orthonormalize(jnp.eye(2)[None], metric)
    change = jnp.array([[.13, -.08], [-.08, -.04]])
    transport = lambda s: orbital._orthonormalize(coeff, s, reference_metric=metric)
    current, tangent = jax.jvp(transport, (metric,), (change,))
    np.testing.assert_array_equal(current, coeff)
    c, dc = current[0], tangent[0]
    constraint = dc.T@metric@c+c.T@change@c+c.T@metric@dc
    np.testing.assert_allclose(constraint, 0., atol=5e-15, rtol=0)
    assert float(jnp.linalg.norm(tangent)) > .01
    # At a mathematically normalized origin both charts have the same first
    # and second response, with no new freeze of the live metric.
    ordinary = lambda s: orbital._orthonormalize(coeff, s)
    np.testing.assert_allclose(tangent, jax.jvp(ordinary, (metric,), (change,))[1],
                               atol=3e-15, rtol=0)
    second = lambda function: jax.jvp(lambda s: jax.jvp(function, (s,), (change,))[1],
                                      (metric,), (change,))[1]
    np.testing.assert_allclose(second(transport), second(ordinary), atol=3e-15, rtol=0)
