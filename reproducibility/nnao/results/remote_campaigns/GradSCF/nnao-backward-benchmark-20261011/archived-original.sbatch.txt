#!/bin/bash
#SBATCH --partition=home
#SBATCH --nodelist=n003
#SBATCH --gres=gpu:V100:1
#SBATCH --cpus-per-task=32
#SBATCH --mem=64G
#SBATCH --time=00:30:00
#SBATCH --job-name=nnao-backward-bench
#SBATCH --output=/home/user_yuanjiao/private/homefile/GradSCF/artifacts/nnao-backward-benchmark-20261011/job-%j.log
set -euo pipefail
export PATH=/home/user_yuanjiao/private/homefile/conda-envs/jax_scf/bin:$PATH
export PYTHONPATH=src:src/gradscf/model/nnao
export OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1 MKL_NUM_THREADS=1
export XLA_PYTHON_CLIENT_PREALLOCATE=false
export JAX_PLATFORMS=cuda,cpu JAX_ENABLE_X64=1
cd /home/user_yuanjiao/private/homefile/GradSCF/releases/nnao-backward-benchmark-20261011
out=/home/user_yuanjiao/private/homefile/GradSCF/artifacts/nnao-backward-benchmark-20261011
lib=src/gradscf/integrals/_native/lib/libgradscf_integrals.so
cp "$lib" "$out/optimized.so"
trap 'cp "$out/optimized.so" "$lib"' EXIT
hostname
nvidia-smi
lscpu
for config in baseline optimized8 optimized16; do
  case "$config" in
    baseline)
      cp /home/user_yuanjiao/private/homefile/GradSCF/artifacts/native-vjp-parallel-20261011/baseline.so "$lib"
      export GRADSCF_NATIVE_THREADS=1 ;;
    optimized8)
      cp "$out/optimized.so" "$lib"
      export GRADSCF_NATIVE_THREADS=8 ;;
    optimized16)
      cp "$out/optimized.so" "$lib"
      export GRADSCF_NATIVE_THREADS=16 ;;
  esac
  python -u tools/benchmark_nnao_backward.py \
    --structures /home/user_yuanjiao/private/homefile/GradSCF/artifacts/nnao-joint200-20261011/train.jsonl \
    --checkpoint /home/user_yuanjiao/private/homefile/GradSCF/artifacts/nnao-joint200-20261011/initial.npz \
    --structure-id S22/19a --epochs 3 --repeats 3 --output-dir "$out/$config"
done
python "$out/compare.py" "$out"
