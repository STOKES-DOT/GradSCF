"""A float64 packed optimizer vector must not unravel to float32 parameters."""
import jax
import jax.numpy as jnp
from jax.flatten_util import ravel_pytree
from flax import nnx
import numpy as np
import pytest


def test_float64_parameters_preserve_initial_values_and_small_updates():
    from gradscf.model.nnao.encoder import _use_float64_parameters
    class Mixed(nnx.Module):
        def __init__(self):
            self.single = nnx.Param(jnp.array([.125], dtype=jnp.float32))
            self.double = nnx.Param(jnp.array([.5], dtype=jnp.float64))
    model = Mixed()
    before, _ = ravel_pytree(nnx.state(model, nnx.Param))
    assert before.dtype == jnp.float64  # This alone did not certify leaf precision.
    _use_float64_parameters(model)
    parameters = nnx.state(model, nnx.Param)
    initial, unravel = ravel_pytree(parameters)
    np.testing.assert_array_equal(initial, before)
    assert all(x.dtype == jnp.float64 for x in jax.tree.leaves(parameters))
    point = initial + 1e-10
    restored, _ = ravel_pytree(unravel(point))
    np.testing.assert_array_equal(restored, point)
    function = lambda x: sum(jnp.sum(leaf) for leaf in jax.tree.leaves(unravel(x)))
    direction = jnp.array([.3, -.4])
    step = 1e-6
    fd = (function(point+step*direction)-function(point-step*direction))/(2*step)
    np.testing.assert_allclose(jax.grad(function)(point)@direction, fd, atol=1e-9, rtol=0)


def test_real_mace_parameter_leaves_and_trained_head_directional_derivative():
    pytest.importorskip('cuequivariance')
    pytest.importorskip('mace_jax')
    from gradscf.model.nnao import MACEBasisModel, build_graph
    from gradscf.model.nnao.config import MODEL_CONFIG
    model = MACEBasisModel(**MODEL_CONFIG, rngs=nnx.Rngs(0))
    definition, parameters, other = nnx.split(model, nnx.Param, ...)
    assert all(x.dtype == jnp.float64 for x in jax.tree.leaves(parameters))
    initial, unravel = ravel_pytree(parameters)
    # Nonzero heads expose the backbone's derivative, hidden at zero init.
    theta = initial+.002*jnp.sin(jnp.arange(initial.size))
    graph = build_graph([8, 1, 1], [[0, 0, 0], [0, -.757, .587], [0, .757, .587]],
                        element_order=MODEL_CONFIG['elements'])
    weight = jnp.cos(jnp.arange(54)).reshape(3, 3, 6)
    f = jax.jit(lambda x: jnp.sum(nnx.merge(definition, unravel(x), other)(graph)*weight))
    direction = np.random.default_rng(0).normal(size=theta.shape)
    direction /= np.linalg.norm(direction)
    ad = jax.grad(f)(theta)@direction
    for step in (1e-3, 1e-4):
        fd = (f(theta+step*direction)-f(theta-step*direction))/(2*step)
        np.testing.assert_allclose(ad, fd, atol=2e-8, rtol=2e-6)
